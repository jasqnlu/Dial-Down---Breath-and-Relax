# 3D Anatomy Import Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the 2D vector-art Muscle/Skeleton Body Map layers with decimated 3D models exported from the Z-Anatomy Blender source files, rendered exactly the way the existing Skin layer already renders its 3D model.

**Architecture:** A one-time Blender headless export pipeline (Python script, not part of the app target) produces two new bundled OBJ resources (`BodyMuscle.obj`, `BodySkeleton.obj`) alongside the existing `BodyMale.obj`. `BodySceneView`/`BodyRig` (currently hardcoded to Skin) become layer-parameterized so all three Body Map layers load through the same SceneKit path. `BodyMapView` drops its Skin-only special case so Muscle/Skeleton get the same free-rotate-when-browsing / locked-with-tap-grid-when-marking behavior Skin already has. The now-dead 2D vector art for Muscle/Skeleton is deleted.

**Tech Stack:** Blender 5.1 (Python API, headless), Swift/SwiftUI, SceneKit, Swift Testing.

## Global Constraints

- Match Skin's existing fidelity: one merged, decimated mesh per layer, ~15,000 triangles (same budget as `BodyMale.obj`), single flat `physicallyBased` material per layer — no per-part mesh identity, no per-muscle/per-bone naming.
- Muscle layer = "Muscular system" + "Muscular insertions" collections only (no skeleton/joints, even though they're also visible in the source `.blend`). Skeleton layer = "Skeletal system" collection only.
- No female-specific 3D variant — one model per layer regardless of the `bodyMapSex` toggle, matching how `BodyMale.obj` already works today.
- Per-layer material colour reuses the exact RGB already defined on `BodyLayer.highlightColor` (`Models/BodyPart.swift` defines `BodyLayer`; the color extension lives in `Views/BodyMap/HumanFigureView.swift`), so the 3D render matches the rest of the UI.
- Source `.blend` files (`~/Blender/Z-AnatomyMuscle.blend`, `~/Blender/Z-AnatomySkeleton.blend`) are opened read-only by the export script and never saved back to disk.
- Z-Anatomy is CC BY-SA 4.0 — ship an in-app Credits entry naming the project and license. Do not link to any Z-Anatomy URL unless the user supplies one directly (don't guess/fabricate a URL).

- [ ] **Step 1: Write the export script**

Create `Scripts/bodymap-import/export_anatomy.py`:

```python
"""Export a merged, decimated OBJ from one or more collections of a
Z-Anatomy .blend file. Run headlessly through Blender:

/Applications/Blender.app/Contents/MacOS/Blender --background \
    "/Users/jasonlu/Blender/Z-AnatomySkeleton.blend" \
    --python Scripts/bodymap-import/export_anatomy.py -- \
    --collections "1: Skeletal system" \
    --out "Breath - Relax & Stretch/Resources/Models3D/BodySkeleton.obj" \
    --target-tris 15000

The source .blend file is opened read-only: this script joins/decimates
objects in the in-memory scene only and never calls save_mainfile(), so the
source file on disk is never modified.

forward_axis='Y', up_axis='Z' were derived empirically (see the plan doc /
export_anatomy.py's originating design notes) from named skeletal landmarks
(Frontal vs Occipital bone, Calcaneus vs Parietal bone) confirming native
Blender space here is Z-up with the body's front toward -Y — which Blender's
axis_conversion (to_forward='-Z', to_up='Y', the fixed OBJ convention) maps
correctly to Y-up / front-facing +Z only when forward_axis='Y' (not '-Y').
This matches the convention BodySceneView.swift already documents for
BodyMale.obj: Y-up, face toward +Z.
"""
import argparse
import sys
import bpy


def parse_args():
    argv = sys.argv
    idx = argv.index("--") if "--" in argv else len(argv)
    parser = argparse.ArgumentParser()
    parser.add_argument("--collections", required=True,
                         help="Comma-separated collection names (visible mesh objects only)")
    parser.add_argument("--out", required=True)
    parser.add_argument("--target-tris", type=int, required=True)
    return parser.parse_args(argv[idx + 1:])


def visible_mesh_objects(collection_names):
    vl = bpy.context.view_layer
    wanted = set(collection_names)
    objs = []
    for obj in bpy.data.objects:
        if obj.type != 'MESH':
            continue
        if not obj.visible_get(view_layer=vl):
            continue
        if any(c.name in wanted for c in obj.users_collection):
            objs.append(obj)
    return objs


def main():
    args = parse_args()
    collection_names = [c.strip() for c in args.collections.split(",")]

    objs = visible_mesh_objects(collection_names)
    if not objs:
        raise RuntimeError(f"No visible mesh objects found in collections {collection_names}")
    print(f"SOURCE_OBJECTS: {len(objs)}")

    bpy.ops.object.select_all(action='DESELECT')
    for obj in objs:
        obj.select_set(True)
    bpy.context.view_layer.objects.active = objs[0]

    bpy.ops.object.join()
    merged = bpy.context.view_layer.objects.active

    total_tris = sum(len(p.vertices) - 2 for p in merged.data.polygons)
    print(f"MERGED_TRIS: {total_tris}")

    if total_tris > args.target_tris:
        ratio = args.target_tris / total_tris
        mod = merged.modifiers.new(name="Decimate", type='DECIMATE')
        mod.decimate_type = 'COLLAPSE'
        mod.ratio = ratio
        mod.use_collapse_triangulate = True
        print(f"DECIMATE_RATIO: {ratio:.4f}")
    else:
        print("DECIMATE_RATIO: skipped (already under target)")

    bpy.ops.object.select_all(action='DESELECT')
    merged.select_set(True)
    bpy.context.view_layer.objects.active = merged

    bpy.ops.wm.obj_export(
        filepath=args.out,
        export_selected_objects=True,
        forward_axis='Y',
        up_axis='Z',
        apply_modifiers=True,
        export_materials=False,
        export_uv=False,
        export_normals=True,
        export_triangulated_mesh=True,
        global_scale=1.0,
    )
    print(f"EXPORTED: {args.out}")


main()
```

Create `Scripts/bodymap-import/verify_obj.py`:

```python
"""Parse a Wavefront OBJ file (no Blender/trimesh dependency) and check its
bounding-box orientation and triangle count.

Usage:
    python3 Scripts/bodymap-import/verify_obj.py path/to/model.obj --min-tris 8000 --max-tris 22000
"""
import argparse
import sys


def parse_obj(path):
    verts = []
    tri_count = 0
    with open(path, "r") as f:
        for line in f:
            if line.startswith("v "):
                parts = line.split()
                verts.append((float(parts[1]), float(parts[2]), float(parts[3])))
            elif line.startswith("f "):
                tri_count += len(line.split()) - 1 - 2
    return verts, tri_count


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("path")
    parser.add_argument("--min-tris", type=int, required=True)
    parser.add_argument("--max-tris", type=int, required=True)
    args = parser.parse_args()

    verts, tri_count = parse_obj(args.path)
    if not verts:
        print(f"FAIL: no vertices found in {args.path}")
        sys.exit(1)

    xs = [v[0] for v in verts]
    ys = [v[1] for v in verts]
    zs = [v[2] for v in verts]
    x_range = max(xs) - min(xs)
    y_range = max(ys) - min(ys)
    z_range = max(zs) - min(zs)

    print(f"VERTS: {len(verts)}  TRIS: {tri_count}")
    print(f"BBOX_RANGE: x={x_range:.4f} y={y_range:.4f} z={z_range:.4f}")

    ok = True
    if not (args.min_tris <= tri_count <= args.max_tris):
        print(f"FAIL: triangle count {tri_count} outside [{args.min_tris}, {args.max_tris}]")
        ok = False
    if not (y_range > x_range and y_range > z_range):
        print("FAIL: Y is not the largest bounding-box dimension — model may not be Y-up")
        ok = False

    if ok:
        print("PASS")
    else:
        sys.exit(1)


main()
```

```swift
final class BodyRig {
    let scene = SCNScene()
    let cameraNode = SCNNode()
    let rigNode = SCNNode()
    let layer: BodyLayer

    private(set) var loadFailed = false

    var committedRotationY: CGFloat = 0

    init(layer: BodyLayer = .skin) {
        self.layer = layer
        setUpCamera()
        setUpLights()
        scene.rootNode.addChildNode(rigNode)
        loadBody()
    }

    static let defaultCameraDistance: CGFloat = 2.28

    private func setUpCamera() {
        let camera = SCNCamera()
        camera.fieldOfView = 50
        camera.zNear = 0.1
        camera.zFar = 20
        cameraNode.camera = camera
        cameraNode.position = SCNVector3(0, 0, Float(BodyRig.defaultCameraDistance))
        scene.rootNode.addChildNode(cameraNode)
    }

    private func setUpLights() {
        let key = SCNLight()
        key.type = .directional
        key.intensity = 1000
        key.color = UIColor.white
        let keyNode = SCNNode()
        keyNode.light = key
        keyNode.position = SCNVector3(3, 4, 5)
        keyNode.look(at: SCNVector3(0, 0, 0))
        scene.rootNode.addChildNode(keyNode)

        let fill = SCNLight()
        fill.type = .directional
        fill.intensity = 380
        fill.color = UIColor.white
        let fillNode = SCNNode()
        fillNode.light = fill
        fillNode.position = SCNVector3(-3, 1, 4)
        fillNode.look(at: SCNVector3(0, 0, 0))
        scene.rootNode.addChildNode(fillNode)

        let rim = SCNLight()
        rim.type = .directional
        rim.intensity = 260
        rim.color = UIColor.white
        let rimNode = SCNNode()
        rimNode.light = rim
        rimNode.position = SCNVector3(0, 2, -5)
        rimNode.look(at: SCNVector3(0, 0, 0))
        scene.rootNode.addChildNode(rimNode)

        let ambient = SCNLight()
        ambient.type = .ambient
        ambient.intensity = 180
        ambient.color = UIColor.white
        let ambientNode = SCNNode()
        ambientNode.light = ambient
        scene.rootNode.addChildNode(ambientNode)
    }

    /// Which bundled OBJ backs each layer's 3D model.
    static func resourceName(for layer: BodyLayer) -> String {
        switch layer {
        case .skin:     return "BodyMale"
        case .muscle:   return "BodyMuscle"
        case .skeleton: return "BodySkeleton"
        }
    }

    /// Flat PBR diffuse colour per layer — reuses the same RGB already used
    /// for that layer's UI accent/highlight (see BodyLayer.highlightColor in
    /// HumanFigureView.swift) so the 3D render matches the rest of the UI.
    static func materialColor(for layer: BodyLayer) -> UIColor {
        switch layer {
        case .skin:     return UIColor(red: 0.89, green: 0.72, blue: 0.62, alpha: 1)
        case .muscle:   return UIColor(red: 0.82, green: 0.14, blue: 0.14, alpha: 1)
        case .skeleton: return UIColor(red: 0.52, green: 0.48, blue: 0.44, alpha: 1)
        }
    }

    /// The processed body mesh per layer, built exactly ONCE per process
    /// (see original note on why this is cached: SwiftUI reconstructs
    /// BodyRig on every BodySceneView re-init). Three independent `static
    /// let`s rather than a dictionary so each layer's mesh is only parsed
    /// the first time THAT layer is actually shown, and so the cache stays
    /// safe to touch from Swift's guaranteed-once static initialization
    /// (no shared mutable state to synchronize).
    private static let skinTemplate: SCNNode?     = makeTemplateBodyNode(for: .skin)
    private static let muscleTemplate: SCNNode?   = makeTemplateBodyNode(for: .muscle)
    private static let skeletonTemplate: SCNNode? = makeTemplateBodyNode(for: .skeleton)

    private static func templateBodyNode(for layer: BodyLayer) -> SCNNode? {
        switch layer {
        case .skin:     return skinTemplate
        case .muscle:   return muscleTemplate
        case .skeleton: return skeletonTemplate
        }
    }

    private static func makeTemplateBodyNode(for layer: BodyLayer) -> SCNNode? {
        guard let url = Bundle.main.url(forResource: resourceName(for: layer), withExtension: "obj"),
              let source = try? SCNScene(url: url, options: [.checkConsistency: true])
        else {
            return nil
        }

        let bodyNode = SCNNode()
        for child in source.rootNode.childNodes {
            bodyNode.addChildNode(child)
        }

        let (bmin, bmax) = bodyNode.boundingBox
        let center = SCNVector3((bmin.x + bmax.x) / 2,
                                 (bmin.y + bmax.y) / 2,
                                 (bmin.z + bmax.z) / 2)
        let height = CGFloat(bmax.y - bmin.y)
        let scale = height > 0 ? Float(2.0 / height) : 1

        bodyNode.pivot = SCNMatrix4MakeTranslation(center.x, center.y, center.z)
        bodyNode.scale = SCNVector3(scale, scale, scale)

        applyMaterial(materialColor(for: layer), to: bodyNode)
        return bodyNode
    }

    private func loadBody() {
        guard let template = BodyRig.templateBodyNode(for: layer) else {
            loadFailed = true
            return
        }
        rigNode.addChildNode(template.clone())
    }

    private static func applyMaterial(_ color: UIColor, to node: SCNNode) {
        let material = SCNMaterial()
        material.lightingModel = .physicallyBased
        material.diffuse.contents  = color
        material.roughness.contents = 0.7
        material.metalness.contents = 0.0
        material.isDoubleSided = true

        if let geometry = node.geometry {
            geometry.materials = [material]
        }
        for child in node.childNodes {
            applyMaterial(color, to: child)
        }
    }

    // MARK: - Rotation
    // (applyDragRotation / commitDragRotation / snap are unchanged — leave as-is)
}
```
